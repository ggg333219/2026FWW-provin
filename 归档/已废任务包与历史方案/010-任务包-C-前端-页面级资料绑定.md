### C-1. 新增 `MaterialSidebar.vue` 左栏组件 ⭐（核心增量）

**目标**：S3 工作台左栏——按"当前选中页面"过滤 + 上传 + 勾选 chunks + 解绑。

**文件**：
- 新增：`frontend/teacher/src/components/teacher/MaterialSidebar.vue`
- 引用现有组件：`EvidenceSidebar.vue`（检索/应用证据功能）、`TeacherUploadModal.vue`（上传 UI）、`S2MaterialWorkspace.vue`（拆解为子分区复用）

**实现要点**（组件结构）：
```vue
<template>
  <aside class="material-sidebar">
    <header>
      <h3>当前页面：P{{ selectedPageOrder }} - {{ selectedPageTitle }}</h3>
      <button @click="triggerAutoBind">自动绑定</button>  <!-- C-3 联调触发点 -->
    </header>
    
    <!-- 分区 1：上传 -->
    <section class="upload-section">
      <TeacherUploadModal @uploaded="handleUploaded" />  <!-- 复用 -->
    </section>
    
    <!-- 分区 2：检索（复用 EvidenceSidebar） -->
    <section class="search-section">
      <EvidenceSidebar
        :scope="'page'"
        :target-node-id="selectedPageId"
        @search="handleSearch"
        @apply="handleApply"
      />
    </section>
    
    <!-- 分区 3：chunks 勾选（NEW） -->
    <section class="chunks-section">
      <h4>可用 chunks ({{ chunks.length }})</h4>
      <ul>
        <li v-for="chunk in chunks" :key="chunk.chunk_id">
          <input type="checkbox" :checked="isBound(chunk.chunk_id)" @change="toggleBinding(chunk.chunk_id)" />
          <span>{{ chunk.text_snippet }}</span>
          <span class="auto-bound-tag" v-if="chunk.auto_bound">自动</span>
        </li>
      </ul>
    </section>
    
    <!-- 分区 4：当前绑定状态 -->
    <section class="binding-section">
      <h4>已绑定 ({{ boundCount }})</h4>
      <button @click="clearBindings" v-if="boundCount">清空</button>
    </section>
  </aside>
</template>

<script setup>
import { computed } from 'vue'
import { useMaterialStore } from '../stores/materialStore'
import EvidenceSidebar from './EvidenceSidebar.vue'
import TeacherUploadModal from './TeacherUploadModal.vue'

const props = defineProps({
  selectedPageId: { type: String, required: true },
  selectedPageOrder: { type: Number, default: 0 },
  selectedPageTitle: { type: String, default: '' }
})

const emit = defineEmits(['auto-bound'])

const materialStore = useMaterialStore()

const chunks = computed(() => materialStore.getPageChunks(props.selectedPageId) || [])
const boundCount = computed(() => materialStore.getPageBinding(props.selectedPageId)?.evidence_refs?.length || 0)
const isBound = (chunkId) => materialStore.getPageBinding(props.selectedPageId)?.evidence_refs?.includes(chunkId) || false

async function triggerAutoBind() {
  const result = await teacherV1Api.outline.autoBindEvidence({
    intent_id: intentStore.intentId,
    outline_id: outlineStore.outline_id
  })
  materialStore.setPageBindingsBulk(result.page_bindings)
  emit('auto-bound', result)
}

async function toggleBinding(chunkId) {
  const binding = materialStore.getPageBinding(props.selectedPageId) || { page_id: props.selectedPageId, evidence_refs: [] }
  if (isBound(chunkId)) {
    binding.evidence_refs = binding.evidence_refs.filter(id => id !== chunkId)
  } else {
    binding.evidence_refs.push(chunkId)
  }
  materialStore.setPageBinding(props.selectedPageId, binding)
}
</script>
```

  - **GREEN**：实现直到通过

**完成标志**：

### C-2. `materialStore.js` 改造：`pageBindings` ⭐

**目标**：保留旧 `evidence_packet` 字段，新增 `pageBindings: Record<page_id, PageEvidenceBinding>`。

**文件**：
- 扩展：`frontend/teacher/src/stores/materialStore.js`
- 扩展测试：`frontend/teacher/src/stores/materialStore.test.js`

**实现要点**：
```javascript
// 新增 state
const pageBindings = ref({})  // Record<page_id, PageEvidenceBinding>

// 新增 getters
const getPageBinding = (pageId) => pageBindings.value[pageId] || null
const getPageChunks = (pageId) => {
  const binding = pageBindings.value[pageId]
  if (!binding) return []
  // chunks 来源：旧 s3EvidencePayload.relevant_chunks 过滤 metadata.page_id === pageId
  return s3EvidencePayload.value?.relevant_chunks?.filter(c => c.metadata?.page_id === pageId) || []
}
const boundCount = computed(() => Object.values(pageBindings.value).reduce((sum, b) => sum + (b.evidence_refs?.length || 0), 0))

// 新增 actions
function setPageBinding(pageId, binding) {
  pageBindings.value = { ...pageBindings.value, [pageId]: binding }
}
function setPageBindingsBulk(bindings) {
  pageBindings.value = { ...pageBindings.value, ...bindings }
}
function clearPageBinding(pageId) {
  const next = { ...pageBindings.value }
  delete next[pageId]
  pageBindings.value = next
}
```

- 单测 `materialStore.test.js` 现有用例保持通过
- 新增用例：
  - RED：`setPageBinding("p1", { evidence_refs: ["c1"] })` → `getPageBinding("p1").evidence_refs` 包含 `"c1"`
  - RED：`clearPageBinding("p1")` → `getPageBinding("p1")` 为 `null`
  - RED：`setPageBindingsBulk({p1: {...}, p2: {...}})` → `Object.keys(pageBindings).length === 2`
  - GREEN：实现直到通过
